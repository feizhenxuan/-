## 一、前端对话页面

**文件**：`src/pages/Chat/index.tsx`
**仓库**：https://code.alipay.com/group-one-learn-test/zzoneticket/blob/sprint_testtickethx_S090011815832_20260730/src/pages/Chat/index.tsx

### 讲解

这是智能购票助手的对话页面，核心功能包括：
- SSE 流式对话（打字机效果）
- 四种卡片渲染（电影/影院/场次/订单）
- 一键复制 AI 回复
- 快捷回复引导
- 位置感知（GPS + IP 双兜底）

**Ant Design 组件使用**：
- Avatar：区分用户/AI 头像
- Input：消息输入框，回车发送，加载中禁用
- Button：发送按钮（loading 防重复）+ 复制按钮
- message：全局提示条

**没有用到的 antd 组件**：Card、List、Spin、Modal、Form 等重型组件，全部用 div + 内联样式手写，更灵活，减少包体积。

```typescript
      if (navigator.clipboard && window.isSecureContext) {
        await navigator.clipboard.writeText(text);

        const textarea = document.createElement('textarea');
        textarea.value = text;

        textarea.style.opacity = '0';

        document.execCommand('copy');
        document.body.removeChild(textarea);

      antMessage.success('已复制');

    if (!input.trim() || isLoading) return;
    const text = input.trim();
    setInput('');

    addMessage({ role: 'user', content: text });
    addMessage({ role: 'assistant', content: '' });
    setLoading(true);

    try {

            setLoading(false);

            <RobotOutlined style={{ fontSize: 32, color: '#C0733A' }} />
            <div style={{ fontSize: 22, fontWeight: 700, color: '#fff' }}>智能购票助手</div>
            <div style={{ fontSize: 14, color: 'var(--text-tertiary)' }}>

            </div>
            {quickReplies.map((q) => (
              <div key={q} onClick={() => setInput(q)} style={{ padding: '10px 20px', borderRadius: 20, cursor: 'pointer' }}>
                {q}
              </div>
            ))}
          </div>
        )}

        {/* 消息列表渲染 */}

            {item.rating > 0 && <div>{Number(item.rating).toFixed(1)} 分</div>}
          </div>
        ))}
      </div>
    );
  }

  // 影院列表卡片
  if (card.card_type === 'cinema_list') {
    return (
      <div>
        {card.items?.map((item: any, i: number) => (
          <div key={i}>
            <div>{item.name}</div>
            <div>{item.address}</div>
          </div>
        ))}
      </div>
    );
  }

  // 场次列表卡片
  if (card.card_type === 'session_list') {
    const statusText = (s: string) => {
      const map: Record<string, string> = { available: '有票', sold_out: '售罄', closed: '已关闭' };
      return map[s] || s;
    };
    return (
      <div>
        {card.items?.map((item: any, i: number) => (
          <div key={i}>
            {item.cinema_name && <div>{item.cinema_name}</div>}
            {item.start_time ? (
              <div>{new Date(item.start_time).toLocaleString('zh-CN', { month: 'numeric', day: 'numeric', hour: '2-digit', minute: '2-digit' })}</div>
            ) : (
              <div>时间待定</div>
            )}
            <div>{statusText(item.status)}</div>
            <div>{item.price != null ? `¥${item.price}` : '价格待定'}</div>
          </div>
        ))}
      </div>
    );
  }

  // 订单列表卡片
  if (card.card_type === 'order_list') {
    return (
      <div>
        {card.items?.map((item: any, i: number) => (
          <div key={i}>
            <div>{item.order_no}</div>
            <div>{item.status} | ¥{item.total_amount}</div>
          </div>
        ))}
      </div>
    );
  }

  return null;
}
```

## 二、前端 SSE API 层

**文件**：`src/services/chat.ts`
**仓库**：https://code.alipay.com/group-one-learn-test/zzoneticket/blob/sprint_testtickethx_S090011815832_20260730/src/services/chat.ts

### 讲解

SSE 流式请求的核心实现：
- 用 `fetch + ReadableStream` 而不是 EventSource，因为 EventSource 只支持 GET，这里需要 POST
- buffer 半包拼接：网络数据可能不完整到达，用 buffer 暂存，按 `\n\n` 分割事件
- Token 兼容 localStorage 和 sessionStorage 两种存储
- 四层错误处理：HTTP 状态码 / 网络错误 / 后端业务错误 / JSON 解析容错

### 完整代码

```typescript
/**
 * 对话 API 层 — SSE 流式请求 + 会话管理
 */

import request from './request';

/** 获取登录 Token，兼容 localStorage 和 sessionStorage */
function getToken(): string | null {
  return (
    localStorage.getItem('access_token') ||   // 用户勾选"记住我"
    sessionStorage.getItem('access_token')     // 用户不勾选"记住我"
  );
}

export const chatApi = {
  /**
   * 发送消息 — SSE 流式返回
   *
   * 核心流程：
   *   1. 检查登录态
   *   2. 拼接 URL 参数（消息 + 会话ID + 位置信息）
   *   3. 发起 POST 请求，获取 ReadableStream
   *   4. buffer 拼接半包数据，按 \n\n 分割 SSE 事件
   *   5. 解析 JSON，回调给页面组件
   */
  send: async (
    content: string,
    sessionId?: string,
    onMessage?: (data: any) => void,
    city?: string,
    lat?: number | null,
    lng?: number | null,
    signal?: AbortSignal | null,
  ) => {
    // 第 1 步：检查登录态
    const token = getToken();
    if (!token) {
      onMessage?.({ type: 'error', content: '请先登录后再使用 AI 助手' });
      return;
    }

    // 第 2 步：拼接 URL 参数
    const params = new URLSearchParams({ content });
    if (sessionId) params.append('session_id', sessionId);
    if (city) params.append('city', city);
    if (lat != null) params.append('lat', String(lat));
    if (lng != null) params.append('lng', String(lng));

    // 第 3 步：发起 POST 请求
    const response = await fetch(`/api/chat/send?${params}`, {
      method: 'POST',
      headers: { Authorization: `Bearer ${token}` },
      signal,
    });

    // HTTP 层错误处理
    if (!response.ok) {
      if (response.status === 401) {
        onMessage?.({ type: 'error', content: '登录已过期，请重新登录' });
      } else {
        onMessage?.({ type: 'error', content: `服务器错误 (${response.status})` });
      }
      return;
    }

    // 第 4 步：获取 ReadableStream，逐段读取
    const reader = response.body?.getReader();
    const decoder = new TextDecoder();
    if (!reader) return;

    let buffer = '';
    try {
      while (true) {
        const { done, value } = await reader.read();
        if (done) break;

        buffer += decoder.decode(value, { stream: true });

        // SSE 事件以双换行分隔
        const events = buffer.split('\n\n');
        buffer = events.pop() || '';  // 保留不完整部分

        for (const event of events) {
          const dataLines = event.split('\n').filter((l) => l.startsWith('data:'));
          if (dataLines.length === 0) continue;
          const data = dataLines.map((l) => l.slice(5).trim()).join('');
          if (data) {
            try {
              const parsed = JSON.parse(data);
              onMessage?.(parsed);
            } catch (e) {
              // JSON 解析失败，静默跳过
            }
          }
        }
      }
    } catch (e: any) {
      // AbortError 是用户主动取消，不算错误
      if (e?.name !== 'AbortError') {
        onMessage?.({ type: 'error', content: '网络连接中断，请检查网络后重试' });
      }
    }
  },

  /** 获取对话历史列表 */
  listSessions: () =>
    request.get('/api/chat/sessions') as Promise<{
      items: Array<{ session_id: string; title: string; preview: string; message_count: number }>;
      total: number;
    }>,

  /** 加载某个对话的完整消息 */
  getSessionMessages: (sessionId: string) =>
    request.get(`/api/chat/sessions/${sessionId}/messages`) as Promise<{
      messages: Array<{ role: string; content: string; cards?: any[] }>;
      session_id: string;
    }>,

  /** 删除某个对话 */
  deleteSession: (sessionId: string) =>
    request.delete(`/api/chat/sessions/${sessionId}`) as Promise<{ message: string }>,
};
```

## 三、前端状态管理

**文件**：`src/stores/chatStore.ts`
**仓库**：https://code.alipay.com/group-one-learn-test/zzoneticket/blob/sprint_testtickethx_S090011815832_20260730/src/stores/chatStore.ts

### 讲解

用 Zustand 做状态管理，比 Redux 轻量，不需要写 action/reducer。

核心亮点是**打字机效果**：
- 后端 SSE 推过来的文本先存到 `typingBuffer`
- 用 `setInterval` 每 20ms 取 2-3 个字符追加到消息上
- `flushTyping()` 确保打字机吐完再设置 loading=false

### 完整代码

```typescript
/**
 * 聊天状态管理 — Zustand Store
 *
 * 核心功能：
 *   1. 消息列表管理
 *   2. 打字机效果（buffer + setInterval 逐字吐出）
 *   3. 卡片数据管理
 *   4. 会话管理
 */

import { create } from 'zustand';

export interface ChatMessage {
  role: 'user' | 'assistant';
  content: string;
  cards?: any[];
}

export interface SessionItem {
  session_id: string;
  title: string;
  preview: string;
  message_count: number;
  created_at?: string;
  updated_at?: string;
}

interface ChatState {
  messages: ChatMessage[];
  sessionId: string | null;
  isLoading: boolean;
  sessions: SessionItem[];
  draftInput: string;
  toolStatus: string | null;
  chatExpanded: boolean;
  rankExpanded: boolean;
  setDraftInput: (text: string) => void;
  setToolStatus: (status: string | null) => void;
  setChatExpanded: (expanded: boolean) => void;
  setRankExpanded: (expanded: boolean) => void;
  addMessage: (msg: ChatMessage) => void;
  updateLastMessage: (content: string, append?: boolean) => void;
  addCard: (card: any) => void;
  updateOrderStatus: (orderId: number, status: string) => void;
  setSessionId: (id: string) => void;
  setLoading: (loading: boolean) => void;
  clearMessages: () => void;
  setSessions: (sessions: SessionItem[]) => void;
  loadHistory: (msgs: ChatMessage[], sid: string) => void;
}

// ===== 打字机效果：模块级别维护 buffer 和 timer =====
let typingBuffer = '';
let typingTimer: ReturnType<typeof setInterval> | null = null;
let typingDoneCallbacks: Array<() => void> = [];

/**
 * 启动打字机
 * 每 20ms 从 buffer 取 2-3 个字符，追加到最后一条 AI 消息
 */
function startTyping(set: any) {
  if (typingTimer) clearInterval(typingTimer);
  typingTimer = setInterval(() => {
    if (typingBuffer.length === 0) {
      if (typingTimer) { clearInterval(typingTimer); typingTimer = null; }
      const callbacks = typingDoneCallbacks;
      typingDoneCallbacks = [];
      callbacks.forEach((cb) => cb());
      return;
    }
    const chunk = typingBuffer.substring(0, Math.min(3, typingBuffer.length));
    typingBuffer = typingBuffer.substring(chunk.length);
    set((state: any) => {
      const messages = [...state.messages];
      const last = messages[messages.length - 1];
      if (last && last.role === 'assistant') {
        messages[messages.length - 1] = { ...last, content: last.content + chunk };
      }
      return { messages };
    });
  }, 20);
}

/** 等打字机吐完 buffer 再执行回调 */
export function flushTyping(callback: () => void) {
  if (typingBuffer.length === 0 && !typingTimer) {
    callback();
  } else {
    typingDoneCallbacks.push(callback);
    if (!typingTimer) startTyping(useChatStore.setState);
  }
}

/** 立即停止打字机（切换会话时调用） */
export function stopTyping() {
  if (typingTimer) { clearInterval(typingTimer); typingTimer = null; }
  typingBuffer = '';
  typingDoneCallbacks = [];
}

export const useChatStore = create<ChatState>((set) => ({
  messages: [],
  sessionId: null,
  isLoading: false,
  sessions: [],
  draftInput: '',
  toolStatus: null,
  chatExpanded: false,
  rankExpanded: false,

  setDraftInput: (text) => set({ draftInput: text }),
  setToolStatus: (status) => set({ toolStatus: status }),
  setChatExpanded: (expanded) => set({ chatExpanded: expanded }),
  setRankExpanded: (expanded) => set({ rankExpanded: expanded }),

  addMessage: (msg) => set((state) => ({ messages: [...state.messages, msg] })),

  /**
   * 更新最后一条消息
   * append=true（text_delta）：攒到打字机 buffer 逐字吐出
   * append=false（text/error）：直接替换，清空 buffer
   */
  updateLastMessage: (content, append = false) => {
    if (append) {
      typingBuffer += content;
      if (!typingTimer) startTyping(set);
    } else {
      typingBuffer = '';
      if (typingTimer) { clearInterval(typingTimer); typingTimer = null; }
      set((state) => {
        const messages = [...state.messages];
        const last = messages[messages.length - 1];
        if (last && last.role === 'assistant') {
          messages[messages.length - 1] = { ...last, content };
        }
        return { messages };
      });
    }
  },

  addCard: (card) =>
    set((state) => {
      const messages = [...state.messages];
      const last = messages[messages.length - 1];
      if (last && last.role === 'assistant') {
        messages[messages.length - 1] = { ...last, cards: [...(last.cards || []), card] };
      }
      return { messages };
    }),

  updateOrderStatus: (orderId: number, status: string) =>
    set((state) => {
      const messages = state.messages.map((msg) => {
        if (!msg.cards) return msg;
        const newCards = msg.cards.map((card: any) => {
          if (card.card_type !== 'order_list') return card;
          const newItems = card.items.map((item: any) =>
            item.id === orderId ? { ...item, status } : item,
          );
          return { ...card, items: newItems };
        });
        return { ...msg, cards: newCards };
      });
      return { messages };
    }),

  setSessionId: (id) => set({ sessionId: id }),
  setLoading: (loading) => set({ isLoading: loading }),

  clearMessages: () => {
    typingBuffer = '';
    if (typingTimer) { clearInterval(typingTimer); typingTimer = null; }
    set({ messages: [], sessionId: null, toolStatus: null });
  },

  setSessions: (sessions) => set({ sessions }),

  loadHistory: (msgs, sid) => {
    typingBuffer = '';
    if (typingTimer) { clearInterval(typingTimer); typingTimer = null; }
    set({ messages: msgs, sessionId: sid });
  },
}));
```

## 五、智能体意图识别

**文件**：`backend/app/core/agents/router_agent.py`
**仓库**：https://code.alipay.com/feizhenxuan.fzx/smart-ticket/blob/dev/yunjian/backend/app/core/agents/router_agent.py

### 讲解

意图识别是多 Agent 系统的调度中心，采用四层策略从快到慢逐级降级：

1. **快速关键词路由**：不调 LLM，毫秒级，覆盖约 60% 请求
2. **指代词解析**：结合上下文列表解析"第一个""最后一部"
3. **LLM 意图识别**：DeepSeek Function Calling，15 秒超时
4. **关键词兜底**：LLM 失败时的最后防线

### 完整代码

## 六、智能体长期记忆

**文件**：`backend/app/services/memory.py`
**仓库**：https://code.alipay.com/feizhenxuan.fzx/smart-ticket/blob/dev/yunjian/backend/app/services/memory.py

### 讲解

三级分层记忆架构，参考人类记忆模型：

1. **ShortTermMemory（瞬时记忆）**：内存中维护 messages 列表，滑动窗口 + 压缩摘要 + 工具调用隔离
2. **MidTermMemory（中期记忆）**：MySQL 持久化，会话历史 + 槽位状态 + 摘要
3. **LongTermMemory（长期记忆）**：ChromaDB 向量检索 + MySQL 知识三元组 + 权重衰减

MemoryManager 统一管理三层记忆，`build_context()` 构建上下文，`settle()` 异步沉淀。

### 完整代码

```python
"""
三级分层记忆架构实现

1. 瞬时上下文记忆（窗口层）— ShortTermMemory
   维护发送给 LLM 的 messages 列表，滑动窗口截断 + 动态摘要压缩。

2. 中期会话记忆（MySQL）— MidTermMemory
   使用 SQLAlchemy 异步 Session 持久化会话历史、槽位状态、摘要。

3. 长期全局记忆 — LongTermMemory
   - 向量子模块：ChromaDB 语义检索
   - 图谱子模块：MySQL 三元组表（KnowledgeTriplet）
   - 衰减机制：定期对权重执行 decay_factor 衰减

4. MemoryManager — 统一管理三层记忆，构建上下文 & 沉淀记忆。
"""

from __future__ import annotations

import uuid
import json
from datetime import datetime
from typing import Any

from sqlalchemy import select, update, delete
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.config import settings
from app.utils.logger import get_logger
from app.services.database import AsyncSessionLocal
from app.models.database import ChatSession, ChatMessage, KnowledgeTriplet, MemoryVector

logger = get_logger(__name__)


# ============================================================
# 1. 瞬时上下文记忆（窗口层）
# ============================================================

class ShortTermMemory:
    """
    瞬时上下文记忆：维护发送给 LLM 的 messages 列表。
    - 滑动窗口截断：超过 max_messages 时保留最近的，早期压缩成摘要。
    - 工具调用隔离：工具返回的数据只保留前 500 字符，避免上下文爆炸。
    """

    def __init__(self, max_messages: int | None = None):
        self.max_messages: int = max_messages or settings.max_history_messages
        self.messages: list[dict[str, Any]] = []
        self._summary: str = ""

    def get_messages(self) -> list[dict[str, Any]]:
        """返回当前 messages 列表（含摘要头部）"""
        result: list[dict[str, Any]] = []
        if self._summary:
            result.append({"role": "system", "content": f"[对话摘要] {self._summary}"})
        result.extend(self.messages)
        return result

    def add_message(self, role: str, content: str, tool_calls: list[dict] | None = None) -> None:
        """添加一条消息，工具返回超过 500 字符时截断"""
        msg: dict[str, Any] = {"role": role, "content": content}
        if tool_calls:
            msg["tool_calls"] = tool_calls
        if role == "tool" and len(content) > 500:
            msg["content"] = content[:500] + "...[截断]"
        self.messages.append(msg)
        if len(self.messages) > self.max_messages:
            self.compress()

    def compress(self) -> None:
        """压缩旧消息：保留最近一半，旧消息拼成摘要"""
        keep_count = self.max_messages // 2
        to_compress = self.messages[:-keep_count] if keep_count > 0 else self.messages
        self.messages = self.messages[-keep_count:] if keep_count > 0 else []

        if not to_compress:
            return

        compressed_parts = [f"[{m.get('role', '')}] {m.get('content', '')[:100]}" for m in to_compress]
        new_summary = " | ".join(compressed_parts)
        self._summary = f"{self._summary} | {new_summary}" if self._summary else new_summary
        if len(self._summary) > 1000:
            self._summary = self._summary[-1000:]

    def reset(self) -> None:
        """重置瞬时记忆"""
        self.messages.clear()
        self._summary = ""


# ============================================================
# 2. 中期会话记忆（MySQL）
# ============================================================

class MidTermMemory:
    """
    中期会话记忆：通过 SQLAlchemy 异步 Session 持久化。
    - 会话创建 / 消息保存 / 历史查询
    - 槽位状态读写（合并写入，不是覆盖）
    - 摘要读写
    """

    async def create_session(self, session_id: str, user_id: int | None = None) -> None:
        """创建会话记录"""
        from datetime import timedelta
        expire_at = datetime.utcnow() + timedelta(minutes=settings.session_timeout_minutes)
        async with AsyncSessionLocal() as session:
            existing = await session.scalar(
                select(ChatSession).where(ChatSession.session_id == session_id)
            )
            if existing is not None:
                return
            new_session = ChatSession(
                session_id=session_id, user_id=user_id,
                slots=json.dumps({}), summary="", expire_at=expire_at,
            )
            session.add(new_session)
            await session.commit()

    async def save_message(self, session_id: str, role: str, content: str,
                           tool_name: str = "", importance: float = 0.5) -> None:
        """保存一条消息到 MySQL"""
        async with AsyncSessionLocal() as session:
            msg = ChatMessage(
                session_id=session_id, role=role, content=content,
                tool_name=tool_name, importance=importance,
            )
            session.add(msg)
            await session.commit()

    async def get_history(self, session_id: str, limit: int | None = None) -> list[ChatMessage]:
        """获取会话历史消息，按时间升序"""
        async with AsyncSessionLocal() as session:
            stmt = select(ChatMessage).where(ChatMessage.session_id == session_id).order_by(ChatMessage.gmt_create.asc())
            if limit is not None:
                stmt = stmt.limit(limit)
            result = await session.execute(stmt)
            return list(result.scalars().all())

    async def get_slots(self, session_id: str) -> dict[str, Any]:
        """获取槽位状态"""
        async with AsyncSessionLocal() as session:
            row = await session.scalar(select(ChatSession).where(ChatSession.session_id == session_id))
            if row is None:
                return {}
            slots_raw = row.slots
            return json.loads(slots_raw) if isinstance(slots_raw, str) and slots_raw else (slots_raw or {})

    async def update_slots(self, session_id: str, slots: dict[str, Any]) -> None:
        """更新槽位状态（合并写入，不是覆盖）"""
        async with AsyncSessionLocal() as session:
            row = await session.scalar(select(ChatSession).where(ChatSession.session_id == session_id))
            if row is None:
                return
            existing = {}
            slots_raw = row.slots
            if isinstance(slots_raw, str):
                existing = json.loads(slots_raw) if slots_raw else {}
            elif isinstance(slots_raw, dict):
                existing = slots_raw
            existing.update(slots)  # 合并，不是替换
            row.slots = json.dumps(existing, ensure_ascii=False)
            await session.commit()

    async def get_summary(self, session_id: str) -> str:
        """获取会话摘要"""
        async with AsyncSessionLocal() as session:
            row = await session.scalar(select(ChatSession).where(ChatSession.session_id == session_id))
            return row.summary if row else ""

    async def update_summary(self, session_id: str, summary: str) -> None:
        """更新会话摘要"""
        async with AsyncSessionLocal() as session:
            row = await session.scalar(select(ChatSession).where(ChatSession.session_id == session_id))
            if row is None:
                return
            row.summary = summary
            await session.commit()

# 3. 长期全局记忆
# ============================================================

class LongTermMemory:
    """
    长期全局记忆：
    - 向量子模块：ChromaDB（可选依赖，导入失败时降级）
    - 图谱子模块：MySQL KnowledgeTriplet 三元组表
    - 衰减机制：对 weight 执行 decay_factor 衰减
    """

    def __init__(self):
        self._chroma_client = None
        self._chroma_collection = None
        self._embed_model = None
        self._vector_available = False

        try:
            import chromadb
            self._chroma_client = chromadb.PersistentClient(path=settings.chroma_persist_dir)
            self._chroma_collection = self._chroma_client.get_or_create_collection(
                name="ticket_memory", metadata={"hnsw:space": "cosine"},
            )
            self._vector_available = True
        except Exception as e:
            logger.warning("long_term_chroma_unavailable", error=str(e))

        self._embed_model = None  # BGE 模型暂时禁用

    def _get_embedding(self, text: str) -> list[float] | None:
        """获取文本的向量表示"""
        if self._embed_model is None:
            return None
        try:
            embedding = self._embed_model.encode(text, normalize_embeddings=True)
            return embedding.tolist()
        except Exception:
            return None

    async def store_vector(self, user_id: int, text: str, metadata: dict | None = None) -> str | None:
        """文本转向量存入 Chroma，同时在 MySQL 记录元数据"""
        if not self._vector_available or self._chroma_collection is None:
            return None
        chroma_id = str(uuid.uuid4())
        embedding = self._get_embedding(text)
        if embedding is None:
            return None
        try:
            self._chroma_collection.add(
                ids=[chroma_id], embeddings=[embedding], documents=[text],
                metadatas=[{"user_id": str(user_id), **(metadata or {})}],
            )
        except Exception:
            return None
        # MySQL 元数据
        try:
            async with AsyncSessionLocal() as session:
                vec = MemoryVector(
                    user_id=user_id, chroma_id=chroma_id,
                    content_summary=text[:500], weight=1.0, decay_factor=0.99,
                )
                session.add(vec)
                await session.commit()
        except Exception:
            pass
        return chroma_id

    async def search_vectors(self, user_id: int, query_text: str, top_n: int = 5) -> list[dict]:
        """语义检索：根据用户最新消息搜索相关记忆"""
        if not self._vector_available or self._chroma_collection is None:
            return []
        query_embedding = self._get_embedding(query_text)
        if query_embedding is None:
            return []
        try:
            results = self._chroma_collection.query(
                query_embeddings=[query_embedding], n_results=top_n,
                where={"user_id": str(user_id)},
            )
        except Exception:
            return []
        output = []
        documents = results.get("documents", [[]])
        distances = results.get("distances", [[]])
        if documents and documents[0]:
            for i, doc in enumerate(documents[0]):
                dist = distances[0][i] if distances and distances[0] and i < len(distances[0]) else 1.0
                output.append({"content": doc, "similarity": 1.0 - dist})
        return output

    async def store_triplet(self, user_id, subject, relation, object,
                            subject_type="", object_type="", weight=1.0):
        """存储知识三元组，相同三元组已存在时取较大权重（强化记忆）"""
        async with AsyncSessionLocal() as session:
            existing = await session.scalar(
                select(KnowledgeTriplet).where(
                    KnowledgeTriplet.user_id == user_id,
                    KnowledgeTriplet.subject == subject,
                    KnowledgeTriplet.relation == relation,
                    KnowledgeTriplet.object == object,
                )
            )
            if existing is not None:
                existing.weight = max(existing.weight, weight)
                existing.updated_at = datetime.utcnow()
            else:
                triplet = KnowledgeTriplet(
                    user_id=user_id, subject=subject, relation=relation,
                    object=object, subject_type=subject_type,
                    object_type=object_type, weight=weight,
                )
                session.add(triplet)
            await session.commit()

    async def get_triplets(self, user_id: int) -> list[KnowledgeTriplet]:
        """获取用户的所有三元组，按权重降序"""
        async with AsyncSessionLocal() as session:
            stmt = select(KnowledgeTriplet).where(
                KnowledgeTriplet.user_id == user_id
            ).order_by(KnowledgeTriplet.weight.desc())
            result = await session.execute(stmt)
            return list(result.scalars().all())

    async def search_by_entity(self, user_id: int, entity: str) -> list[KnowledgeTriplet]:
        """模糊匹配实体：subject 或 object 包含指定关键词"""
        async with AsyncSessionLocal() as session:
            stmt = select(KnowledgeTriplet).where(
                KnowledgeTriplet.user_id == user_id,
                (KnowledgeTriplet.subject.like(f"%{entity}%"))
                | (KnowledgeTriplet.object.like(f"%{entity}%"))
            ).order_by(KnowledgeTriplet.weight.desc())
            result = await session.execute(stmt)
            return list(result.scalars().all())

    async def decay_weights(self) -> None:
        """对所有记忆执行权重衰减，模拟人类遗忘曲线"""
        try:
            async with AsyncSessionLocal() as session:
                vectors = await session.execute(select(MemoryVector))
                for vec in vectors.scalars().all():
                    vec.weight *= vec.decay_factor
                triplets = await session.execute(select(KnowledgeTriplet))
                for tri in triplets.scalars().all():
                    tri.weight *= 0.99
                await session.commit()
        except Exception as e:
            logger.warning("long_term_decay_failed", error=str(e))

# 4. MemoryManager — 统一管理三层记忆
# ============================================================

class MemoryManager:
    """
    统一管理三层记忆：
    - build_context(): 构建完整的 LLM 上下文
    - settle(): 异步沉淀记忆（保存消息、提取实体、存向量、衰减）
    """

    def __init__(self):
        self.short_term = ShortTermMemory()
        self.mid_term = MidTermMemory()
        self.long_term = LongTermMemory()

    async def build_context(self, session_id: str, user_id: int | None = None) -> list[dict]:
        """
        构建完整的上下文消息列表：
        a) 从图谱加载用户核心事实
        b) 从向量库检索相关记忆
        c) 从 MySQL 加载会话历史
        d) 融合到瞬时上下文的 messages 中
        """
        context_parts = []

        # a) 图谱核心事实（权重最高的 10 条三元组）
        if user_id is not None:
            try:
                triplets = await self.long_term.get_triplets(user_id)
                if triplets:
                    facts = [f"{t.subject} {t.relation} {t.object}" for t in triplets[:10]]
                    context_parts.append("[用户核心事实] " + "; ".join(facts))
            except Exception:
                pass

        # b) 向量库检索（基于最近一条用户消息，top-3）
        recent_messages = self.short_term.messages
        last_user_msg = ""
        for msg in reversed(recent_messages):
            if msg.get("role") == "user":
                last_user_msg = msg.get("content", "")
                break
        if user_id is not None and last_user_msg:
            try:
                vector_results = await self.long_term.search_vectors(user_id, last_user_msg, top_n=3)
                if vector_results:
                    memories = [r["content"][:100] for r in vector_results]
                    context_parts.append("[相关记忆] " + " | ".join(memories))
            except Exception:
                pass

        # c) MySQL 会话历史（瞬时记忆为空时恢复）
        try:
            history = await self.mid_term.get_history(session_id, limit=self.short_term.max_messages)
            if history and not recent_messages:
                self.short_term.reset()
                for msg in history:
                    self.short_term.add_message(msg.role, msg.content)
        except Exception:
            pass

        # d) 将上下文信息注入为 system 消息
        messages = self.short_term.get_messages()
        if context_parts:
            system_context = "\n".join(context_parts)
            messages = [{"role": "system", "content": system_context}] + messages
        return messages

    async def settle(self, session_id: str, user_id: int | None,
                     user_message: str, assistant_message: str) -> None:
        """
        异步沉淀记忆：
        a) 保存消息到 MySQL
        b) 提取实体更新图谱
        c) 高价值文本存入向量库
        d) 低价值记忆衰减
        """
        # a) 保存消息
        try:
            await self.mid_term.save_message(session_id, "user", user_message, importance=0.7)
            await self.mid_term.save_message(session_id, "assistant", assistant_message, importance=0.6)
        except Exception:
            pass

        # b) 提取实体更新图谱
        if user_id is not None:
            try:
                await self._extract_and_store_triplets(user_id, user_message, assistant_message)
            except Exception:
                pass

        # c) 高价值文本存入向量库
        if user_id is not None:
            try:
                combined_text = f"用户: {user_message}\n助手: {assistant_message}"
                if len(combined_text) > 20:
                    await self.long_term.store_vector(user_id, combined_text,
                        metadata={"session_id": session_id})
            except Exception:
                pass

        # d) 低价值记忆衰减
        try:
            await self.long_term.decay_weights()
        except Exception:
            pass

    async def _extract_and_store_triplets(self, user_id: int, user_message: str, assistant_message: str) -> None:
        """简化版实体提取：基于关键词规则从对话中提取三元组"""
        relation_patterns = [
            ("想看", "想看"), ("喜欢", "喜欢"), ("买了", "购买"),
            ("订了", "购买"), ("推荐", "推荐"),
        ]
        for keyword, relation in relation_patterns:
            if keyword in user_message:
                idx = user_message.index(keyword)
                after = user_message[idx + len(keyword):].strip()
                for prefix in ["了", "的", "电影", "这部电影"]:
                    if after.startswith(prefix):
                        after = after[len(prefix):].strip()
                if after and len(after) <= 100:
                    await self.long_term.store_triplet(
                        user_id=user_id, subject=f"用户{user_id}",
                        relation=relation, object=after,
                        subject_type="user", object_type="movie", weight=0.8,
                    )

        if "推荐" in assistant_message or "建议" in assistant_message:
            snippet = assistant_message[:50]
            await self.long_term.store_triplet(
                user_id=user_id, subject=f"用户{user_id}",
                relation="收到推荐", object=snippet,
                subject_type="user", object_type="recommendation", weight=0.6,
            )
```
